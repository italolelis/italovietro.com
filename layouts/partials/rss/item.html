{{- /* Overrides the theme's RSS item so a link post points at the piece it links to. Off-site writing lives in the archive as pages with `render: never`, which means they have no page of their own -- and .Permalink for such a page resolves to nothing useful. A subscriber would have got an item with an empty link. Pointing and at externalUrl is also the right behaviour on its own terms: a reader who clicks a link post in a feed wants the piece, not a stub. Posts written here are unaffected -- they have no externalUrl, so the default is their own permalink. */ -}} {{- $params := .Page.Params | merge .Site.Params.Page | merge (dict "author" .Site.Params.Author.name) -}} {{- .Page.Title -}} {{- .Page.Params.externalUrl | default .Page.Permalink -}} {{- .Page.Date.Format "Mon, 02 Jan 2006 15:04:05 -0700" -}} {{- $params.author | default (T "author") -}} {{- .Page.Params.externalUrl | default .Page.Permalink -}} {{- " {{- end -}} {{- $content := .Page.Description -}} {{- if $params.rssFullText -}} {{- $content = dict "Content" .Page.Content "Ruby" $params.ruby "Fraction" $params.fraction "Fontawesome" $params.fontawesome | partial "function/content.html" -}} {{- else -}} {{- with .Page.Summary -}} {{- $content = dict "Content" . "Ruby" $params.ruby "Fraction" $params.fraction "Fontawesome" $params.fontawesome | partial "function/content.html" -}} {{- end -}} {{- end -}} {{- $content | replaceRE `]*>.*` "" | replaceRE `]*( /)?>` "" | safeHTML -}} {{- "]]>" | safeHTML -}}
