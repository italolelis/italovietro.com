{{- /* Overrides the theme's footer to carry the social links. They used to be an icon row on the home page, directly under the tagline -- the first interactive thing a visitor met, before anything the site is actually about. The survey in .planning/research/minimal-personal-site-patterns.md counted iconised social rows on 1 of 14 comparable sites; 7 of 14 have no social links on their home page at all, and the ones that keep them use plain text, in the footer, the way danluu.com does. So: text, not icons. Every page, not just the home page. Below the fold, where someone who wants to find you elsewhere will look for them, and out of the way of someone who does not. URLs are built from the same [params.social] values the theme's icon partial reads, so there is one place to change a handle. */ -}} {{- if ne .Site.Params.footer.enable false -}}
{{- /* Custom Content */ -}} {{- with .Site.Params.footer.custom -}}
{{- safeHTML . -}}
{{- end -}} {{- /* Elsewhere */ -}} {{- $social := .Site.Params.social | default dict -}} {{- $links := slice -}} {{- with $social.GitHub -}} {{- $links = $links | append (printf `GitHub` .) -}} {{- end -}} {{- with $social.Linkedin -}} {{- $links = $links | append (printf `LinkedIn` .) -}} {{- end -}} {{- if $social.RSS -}} {{- $links = $links | append (printf `RSS` ("index.xml" | relLangURL)) -}} {{- end -}} {{- with $links -}}
{{- delimit . " · " | safeHTML -}}
{{- end -}} {{- /* Hugo and LoveIt */ -}} {{- if ne .Site.Params.footer.hugo false -}}
{{- $hugo := printf `Hugo` hugo.Version -}} {{- $theme := .Scratch.Get "version" | printf ` LoveIt` -}} {{- dict "Hugo" $hugo "Theme" $theme | T "poweredBySome" | safeHTML }}
{{- end -}}

{{ . | safeHTML }} {{- end -}}
{{- end -}}
