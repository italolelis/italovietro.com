{{- /* layouts/shortcodes/talk.html */ -}} {{- /* Parameters: title (required), event (required), date (required), type (required), video_url (optional), slides_url (optional), recordings (optional), event_url (optional), highlights_url (optional), highlights_label (optional). type is talk | panel | podcast | host. `event_url` is the panel case. A panel usually leaves no recording and no slides, but the event that hosted it has a page -- so without this a panel entry is the only kind with nothing to click, which reads as less real than the talks above it. `recordings` is for a talk given more than once: pipe-separated `Label=URL` pairs, rendered as one link per label. One entry with three recordings rather than three entries with the same title -- the talk is the thing, and repeating it three times down the page would say the opposite of what it means, which is that a conference asked for it again. Labels are written per language in the content file, so a city can be Florence in one and Florença in the other. `highlights_url` is an episode with a page of its own here (content/episodes/): chapter by chapter, with the lessons pulled out, written by Claude from the transcript and saying so. Labelled "Highlights" rather than "Read" because it is not Italo's writing. It is the one link on the row that stays on this site, so it comes first. Its word defaults to the language's own (i18n `entry.highlights`); `highlights_label` overrides it, which the content still does. The words of the other links (Watch, Slides, Event) are i18n keys too, `entry.watch` and so on. They were English in the template, so the Portuguese page said "Watch" and "Slides" between Portuguese cities. A recording's own label (`recordings`) is written per language in the content, as before. This renders an Entry (layouts/partials/entry.html), the same as the reading list's and the archive's. What is left here is what only a talk knows: which links it has, in what order, and what to call them. */ -}} {{- $title := .Get "title" -}} {{- $event := .Get "event" -}} {{- $date := .Get "date" -}} {{- $type := .Get "type" | default "talk" -}} {{- $videoUrl := .Get "video_url" | default "" -}} {{- $slidesUrl := .Get "slides_url" | default "" -}} {{- $recordings := .Get "recordings" | default "" -}} {{- $eventUrl := .Get "event_url" | default "" -}} {{- $highlightsUrl := .Get "highlights_url" | default "" -}} {{- $highlightsLabel := .Get "highlights_label" | default (i18n "entry.highlights") -}} {{- $inner := trim .Inner "\n" | .Page.RenderString -}} {{- /* The icon is the partial's, picked by kind: talk, panel, podcast, host. */ -}} {{- /* Collected into one slice of { label, url } and handed over, rather than emitted inline with a separator before each optional part. Four optional kinds of link means the inline form needs a condition per boundary, and an earlier version had already lost one: a talk with both `video_url` and `recordings` ran "Watch" straight into the first city with no separator. A list cannot have that bug, whatever combination an entry carries; the partial joins it. */ -}} {{- $links := slice -}} {{- with $highlightsUrl }}{{- $links = $links | append (dict "label" $highlightsLabel "url" (relLangURL .)) -}}{{- end }} {{- with $videoUrl }}{{- $links = $links | append (dict "label" (i18n "entry.watch") "url" .) -}}{{- end }} {{- with $recordings }} {{- range split . "|" }} {{- $parts := split . "=" -}} {{- $links = $links | append (dict "label" (index $parts 0) "url" (index $parts 1)) -}} {{- end }} {{- end }} {{- with $slidesUrl }}{{- $links = $links | append (dict "label" (i18n "entry.slides") "url" .) -}}{{- end }} {{- with $eventUrl }}{{- $links = $links | append (dict "label" (i18n "entry.event") "url" .) -}}{{- end }} {{- /* The title is plain text, not a link: a talk's destinations are on the meta line. An h3 under the group's h2, as the reading list's are; an h4 skipped a level and left the outline broken. */ -}} {{- partial "entry.html" (dict "title" $title "meta" $event "links" $links "date" $date "note" $inner "kind" $type) -}}
