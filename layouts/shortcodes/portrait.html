{{- /* layouts/shortcodes/portrait.html */ -}} {{- /* The portrait, at more than one resolution. It lives here rather than on the home page. Only 2 of the 14 sites surveyed in .planning/research/minimal-personal-site-patterns.md show a face on their home page, and both belong to people with something to sell. On an About page a portrait is the point. Event organisers ask for a headshot by email, so sizes are offered as links rather than left inside a srcset only a browser can read: a 512px square, and the full-size photograph. The source is the speaker kit's headshot (content/speaker-kit/headshot.jpg, 1416px), so About and the kit show one photograph and cannot drift apart; a new headshot dropped into the kit changes both. It was a 512px JPEG of its own, and before that a 428KB PNG of the same 512x512 image. The page itself loads only square crops cut from it here, 256px and 512px for the 176px circle at 1x and 2x, so the full 370KB file is fetched only by someone who clicks for it. Fill, not Resize: the source is not quite square, and Resize to 256x256 would squash it. Parameters: alt (required), caption (required -- the label before the size links, so it can be translated by the caller). */ -}} {{- $kit := site.GetPage "/speaker-kit" -}} {{- $src := cond (ne $kit nil) ($kit.Resources.Get "headshot.jpg") nil -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- if not $alt -}} {{- errorf "portrait.html: alt is required -- a portrait with no alt text tells a screen reader nothing about who is on the page" -}} {{- end -}} {{- if $src -}} {{- $small := $src.Fill "256x256 jpg q88 Center" -}} {{- $medium := $src.Fill "512x512 jpg q88 Center" -}} {{- else -}} {{- errorf "portrait.html: the speaker kit's headshot (content/speaker-kit/headshot.jpg) is missing" -}} {{- end -}}
