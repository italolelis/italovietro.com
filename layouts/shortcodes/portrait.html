{{- /* layouts/shortcodes/portrait.html */ -}} {{- /* The portrait, at more than one resolution. It lives here rather than on the home page. Only 2 of the 14 sites surveyed in .planning/research/minimal-personal-site-patterns.md show a face on their home page, and both belong to people with something to sell. On an About page a portrait is the point. Event organisers ask for a headshot by email today, so both sizes are offered as links rather than left inside a srcset only a browser can read. 512px is the full extent of the original -- there is no larger source to offer, and upscaling would hand over a softer image than the one asked for. Sizes come from Hugo's image processing, so the two files are derived from one source and cannot drift apart. The source is a 59KB JPEG; it was a 428KB PNG of the same 512x512 photograph, which is 7x the bytes for no extra pixels. Parameters: alt (required), caption (required -- the label before the size links, so it can be translated by the caller). */ -}} {{- $src := resources.Get "images/portrait.jpg" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- if not $alt -}} {{- errorf "portrait.html: alt is required -- a portrait with no alt text tells a screen reader nothing about who is on the page" -}} {{- end -}} {{- if $src -}} {{- $small := $src.Resize "256x256 jpg q88" -}} {{- else -}} {{- errorf "portrait.html: assets/images/portrait.jpg is missing" -}} {{- end -}}
