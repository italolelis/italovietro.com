{{- /* layouts/shortcodes/upcoming.html */ -}} {{- /* The next appearances, from data/upcoming.yaml. Renders nothing -- not even the heading -- once every entry's `until` date has passed. An empty "Upcoming" heading says the opposite of what it is there to say, and a page that lists a conference three months after it happened is worse than a page that never claimed to look forward at all. Entries are dropped by date rather than by anyone remembering to delete them, because the failure mode is silent: nobody visits their own speaking page the week after a talk to check it aged correctly. Each one is an Entry (layouts/partials/entry.html), the same as the talks below it, so the group sits in the same visual language by construction and not by a copy of the markup, which is what this used to be. Parameters: heading (required -- passed per language, since the data file holds venues and dates rather than prose). */ -}} {{- $heading := .Get "heading" -}} {{- $future := slice -}} {{- range hugo.Data.upcoming -}} {{- if (time.AsTime .until).After now -}} {{- $future = $future | append . -}} {{- end -}} {{- end -}} {{- with $future }}

{{ $heading }}

{{- range . }} {{- /* The title links to the event when there is a page for it, and the Entry (the partial) opens it in a new tab. The calendar icon is the partial's, picked by the `upcoming` kind. */ -}} {{- partial "entry.html" (dict "title" .event "link" .url "meta" .venue "date" .display "kind" "upcoming") -}} {{- end }} {{- end -}}
